# OutRunners (MAME set 'orunners', World) - V70 program symbols
#
# Addresses and names only: no game data lives in this file.
# Format:  ADDRESS  KIND  NAME  [LAYOUT]   [# comment]
#   KIND: func = function entry, code = code label, data = data label
#   A code label of a computed call or jump may say where it goes: calls=TABLE, the code of a described table
#   register RN NAME: a register the whole program shares, a global variable of the C (its name)
# Addresses use the copy of the program the code refers to:
#   000000-03FFFF (ic37 + ic40, a 16-bit word of each in every 32-bit word), 100000-1FFFFF (ic36 + ic39)
# Mirrors are accepted and folded (e.g. FFFFF0 -> 03FFF0).
#
# LAYOUT, for data tables that hold code addresses the analysis cannot see the code load:
#   record=N        the table is an array of N-byte records
#   count=N         ... of N records, or
#   count=prefix    ... of as many records as the 16-bit word before them says, or
#   count=prefix32  ... of as many records as the 32-bit word before them says, or
#   count=bytes32   ... of as many records as fill the length in bytes the 32-bit word before them says, or
#   end=XXXX        ... closed by this 16-bit word where the next record would start
#   kinds=O:K=N,..  ... of records whose length the 16-bit field at offset O picks: N bytes for kind K
#                   (hexadecimal), 'end' for a kind that closes the array, or 'fork' for one that closes it
#                   with a table of addresses of arrays that go on from there (as many as fit before the
#                   first of them)
#   code=A,B        every record holds a 32-bit code address at these offsets (0 = no code);
#                   A:BASE is a 16-bit offset from BASE instead
#   data=A,B        every record holds at these offsets a 32-bit value that its code may take as the address
#                   of data: the data it goes with, when the value's low 24 bits (the V70's address bus) are
#                   an address of the program ROM that is not code (0 and numbers are not)
#   lists=N         the table is N 16-bit offsets, from its own address, to such arrays
#   pointers=N      the table is N 32-bit addresses of such arrays (0: no array)
# Sizes, offsets and counts are decimal; end and BASE are hexadecimal, like the data and addresses.

# ======== entry points ========
03FFF0  func  reset              # V70 reset vector (0xFFFFFFF0 on a 24-bit bus): "jmp 6"

03F000  data  vector_table       # 256 x 32-bit entries, base loaded into SBR by boot
03F8B4  func  int_default        # target of 250 of the 256 vector table entries

# The V70 takes maskable interrupts at vector 0x40 + the number the interrupt controller (D00000) supplies

03F794  func  int_vector_41
03F7A0  func  int_vector_42
03F7AC  func  int_vector_43

# ======== the registers the program shares ========
# Globals of the C (decompiler/signatures.py CONTEXT); each task has its own copy, which the runtime swaps
# with it. R12, R14, R26, R27 and R28 are shared too, as scratch registers whose last value the next
# function may read (a handler what the one before it left), and keep their register names
register R20  current_object     # the object the object loop (011CC0) runs: set before it calls the handler at field 4
register R23  player_car         # the screen's car: the object at 1000[R25] (209000 or 219000)
register R25  screen             # the block of the screen the task runs: 208000 for screen 1 (the second task), 218000 for screen 2 (the boot task)
register R32  stack_pointer      # SP, in the functions whose stack stays in memory (decompiler/stack.py); 21F400 after boot

# ======== structures ========
# The game's own (decompiler/pointers.py works out which registers point at them): the objects, 124 in the
# page of each screen's block, run by the object loop; the screen's car, at 1000 of its block (an object
# in its first 100 bytes); the block of the screen a task runs (R25)
struct object 100
struct car 200
struct screen 8000
# Each player's race record, two copies of 0x80 bytes in the game's variables: 20FB00 and 20FB80 (stage at
# 60, node at 62, the race time at 68), and 20FC00 and 20FC80, which the time is copied to
struct race 80
field object 00 flags u8               # bit 0: the object is live (run_objects runs it; else it frees its slot)
field object 04 handler u32            # the code run_objects runs for it each frame ("jsr [4[R20]]", 011CD2)
field object 0C state u8               # the step its handler is at: most handlers go by it through a table
field screen 20 race_copy u32 race     # 20FC00 for screen 1, 20FC80 for screen 2
field screen 24 race u32 race          # 20FB00 for screen 1, 20FB80 for screen 2
field car 0 object u32 object          # the car's object, in the screen's objects (set as it starts, 0072xx)

# ======== the frame loop ========
# 0008F0-000912: the program waits for the next frame polling 23[R24] (20E023), which an interrupt sets,
# and clears it; while B00[R25] says data is being unpacked it calls 010E42 for a piece of it each time
000908  code  frame_wait_loop    # "test.b 23[R24]": the poll
00090D  code  frame_wait_done    # the wait is over: a frame is done

# ======== the tasks ========
# The program runs two tasks, each on a stack of its own: the one boot starts runs the loop of screen 2
# (R25 = 218000, 000719), the other the loop of screen 1 (R25 = 208000, 0006B0). Boot (0005C5-0005EF)
# copies the second one's first interrupt frame (registers, PC 0006B0, PSW 00040000) from 000C98 to
# 21FF84, and keeps its stack pointers at 0[R24] (the frame) and 4[R24] (21F800, its L0SP). When 24[R24]
# is clear, the VBLANK handler swaps them (03F5FC): it stores the running task's pair there and returns
# into the other one
0006B0  func  second_task        # where the second task's first frame returns to
task  second_task
03F5FC  code  task_swap          # 03F5FC-03F608: the running task's stacks go to 0[R24] and 4[R24], the other one's come back
03F60C  code  task_swap_done     # the handler goes on with the other task's interrupt frame: it returns into that task

# ======== the objects ========
# Below each screen's block (screen - 8000): this frame's list of objects at -8000 (a count, then the low
# halves of their addresses), the next frame's at -7FF8 (the two lists, at -7E00 and -7D00, swap every
# frame), the used slots at -7FF0 (7C bytes, FF used), a spare object at -7F00 and the 124 objects from
# -7C00, 100 bytes each
011CC0  func  run_objects        # the object loop: each live object of this frame's list (flags bit 0) has its handler run and goes on the next frame's list; a dead one frees its slot; then this frame's list is cleared and the two swap
011D43  func  start_object       # the first free slot: its 100 bytes cleared, live (flags bit 0), its slot number at 8, on this frame's list; R21 = the object (the spare at screen - 7F00 when no slot is free)

# The objects of each stage's road: a stream of (position, command) halfword pairs, taken from this table by
# the stage's number (012720 and 0127BF, "mov.w 20800(R0), 80[R20]") and read at 01272C-012746 as the road
# goes by. Command 0 starts an object (handler -> field 4, then a parameter -> field 18, 012758); 1 ends the
# stage's stream, which goes on from this table when the next stage starts (012777, 0127A4-0127CA); 2 ends
# it for good (01279C); 3 is a fork: a table of the streams of its branches follows, and the one the route
# takes, C9[R23], goes on when the stage changes (012786, 0127CD-0127EB)
020800  data  road_object_streams  pointers=32 record=12 kinds=2:0=12,1=end,2=end,3=fork code=4 data=8

comment scenery_update Static scenery along the road (trees, buildings, crowds, banners, signs): a row of identical sprites from its record. By state: 0 road_row_start; 1 road_row_crash_check, then road_row_draw; 2 (a record without a hit box) road_row_draw only. Ends when road_row_draw finds the whole row behind the view.
comment scenery_pair_update Static scenery drawn twice, at each side of the road, the second mirrored (rows of trees, houses, spectators). By state: 0 road_row_start; 1 road_row_pair_crash_check, then road_row_pair_draw; 2 (no hit box) road_row_pair_draw only.
comment fork_sign_update The signs on the road where it forks at the end of a stage: a line down the middle before the fork and a line across each branch after it. By state: 0 road_row_start; 1 fork_sign_hit_check, then road_row_draw; 2 road_row_draw only. A car that hits one is slowed, not crashed.
comment fork_sign_hit_check road_row_hit_check with kind 3 (R27): a car that hits a fork sign is slowed for 15 frames, as off the road (car unk_147), instead of crashing.
comment animated_scenery_update Scenery that animates in place (the Ferris wheel, windmills; also the scenery of the goal lines). By state: 0 road_row_start_copy; 1 road_row_crash_check; 1 and 2 animation_advance and road_row_draw.
comment animated_scenery_pair_update Animated scenery drawn twice, at each side of the road, the second mirrored (the people lining the start and the goal, the scenery around the forks). By state: 0 road_row_start_copy; 1 road_row_pair_crash_check; 1 and 2 animation_advance and road_row_pair_draw.

By state: 0 screen_centered = FF, then road_row_start (its records have no hit box); 1 and 2 animation_advance and road_row_draw.

comment road_row_draw Draws the row's items in view: ends the object once end_position is not past the view's start (race_copy unk_46); nothing while first_position is beyond the draw distance (SCREEN(screen)->unk_12 >> 7 segments); moves first_position up to the first item in view; for each item (spacing apart, to the draw distance) where the road has a scale: x = (the road's width * side + x_offset) * scale >> 12 plus the road's screen x unless screen_centered, y = y_offset * scale >> 12 plus the road's screen y; off the screen (x beyond -1E0..1E0, y beyond -160..80) it is skipped, else the scale shifted by scale_shift picks the size (size_table: a size step and a shift) and a sprite of the animation's frame with its palette (flags bits 6-7 flip it) goes into the screen's sprite list (unk_C0; the copy at +3000 is the halfway point from the item's sprite of the frame before, previous_sprites) and the depth list of its road segment (screen + 5000). comment road_row_pair_draw road_row_draw for the rows drawn in mirrored pairs: each item twice, at the same distance either side of the road's centre, the second with sprite attribute bits 43 toggled (bit 6 flips it, bits 0-1 swap its X alignment); there is no off-screen test and no screen_centered. Used by scenery_pair and animated_scenery_pair.
comment animation_start Starts animation R9 (rom_034138[R9] is its record) at its first frame: frame_time and frame_countdown from the record's first byte, size_step 0, and fields 34 and 36 to 100 (the width and height zoom of the screen-position drawing, sub_00EF32: 100 the sprite's own size).
